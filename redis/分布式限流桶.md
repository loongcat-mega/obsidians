```lua
-- 该文件只做记录方便编写脚本， 实际执行代码在 //video_fingerprint/common/singleton.cc #kRedisLimitScript 变量中  
-- 如果有更新在本文件中更新不生效  
local tokens_key = KEYS[1]  
local timestamp_key = KEYS[2]  
local timestamp_val = KEYS[3]  
local rate = tonumber(ARGV[1])  
local capacity = tonumber(ARGV[2])  
local now = tonumber(ARGV[3])  
local requested = tonumber(ARGV[4])  
-- 每次接受到请求时，先判断当前 key 是否存在，当前 key 也就是每 100 ms的时间戳，  
local set_result = redis.call("SET", timestamp_val, 1, "EX", 1, "NX")  
if string.lower(set_result) == "OK"  
then  
    -- 令牌填充满需要的时间  
    local fill_time = capacity / rate  
    -- TTL 设置为两倍的填充时间  
    local ttl = math.floor(fill_time * 2)  
    -- 获取上次请求之后剩余的 tokens    local last_tokens = tonumber(redis.call("get", tokens_key))  
    -- 如果上次没有请求 或者已经 expire    -- 则设置 last_tokens 为 capacity，标识‘我’正在用这个 tokens    if last_tokens == nil then  
        last_tokens = capacity  
    end  
    -- 获取上次请求的时间  
    local last_refreshed = tonumber(redis.call("get", timestamp_key))  
    if last_refreshed == nil then  
        last_refreshed = 0  
    end  
    -- 计算两次请求之间的时间差值  
    local delta = math.max(0, now - last_refreshed)  
    -- 从上次请求到现在，应该补充的 tokens 数量+剩余的数量 即目前剩余的 tokens    local filled_tokens = math.min(capacity, last_tokens + (delta * rate))  
    -- allowed：当前的 tokens 是否比请求的 tokens 要多  
    local allowed = filled_tokens >= requested  
    local new_tokens = filled_tokens  
    local allowed_num = 0  
    -- 如果剩余的比请求的多，则设置 new_tokens 为请求之后，桶内剩余的 tokens    if allowed then  
        new_tokens = filled_tokens - requested  
        allowed_num = 1  
    end  
    -- 更新请求时间和桶内剩余 tokens    redis.call("setex", tokens_key, ttl, new_tokens)  
    redis.call("setex", timestamp_key, ttl, now)  
    return allowed_num  
  
else  
    return 0  
end
```
